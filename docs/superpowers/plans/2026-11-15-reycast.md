# ReyCast Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Raycast-style app launcher for the Galaxy Tab S8 Ultra, opened with Cmd+Space through Key Mapper, that fuzzy-matches installed apps, learns usage, and launches on Enter.

**Architecture:** One Kotlin Android app (`dev.victorstein.reycast`): an invisible trampoline activity toggles a `TYPE_APPLICATION_OVERLAY` panel. Pure-Kotlin core (`TextNormalizer`, `Matcher`, `UsageHistory`, `Ranker`, `CatalogCache`, `KeyRouter`) is unit-tested on the Mac; thin Android wrappers (`AppCatalog`, `UsageStore`, `ReyCastOverlay`, `OverlaySession`) are verified on the tablet over adb. A step-0 spike confirms the floating window and the Key Mapper hotkey work before feature code.

**Tech Stack:** Kotlin 2.2, Android Gradle Plugin 8.11, Gradle 8.14, Android SDK 36 (Android 16), plain Android Views (no AndroidX), JUnit 4. Spec: `docs/superpowers/specs/2026-11-15-reycast-design.md`.

```kotlin
package dev.victorstein.reycast

import android.app.Activity
import android.content.Intent
import android.os.Bundle
import android.view.Gravity
import android.view.KeyEvent
import android.view.WindowManager

class LauncherActivity : Activity() {
    private var hasHadFocus = false

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        overrideActivityTransition(OVERRIDE_TRANSITION_OPEN, 0, 0)
        setContentView(R.layout.activity_launcher)
        placeWindow()
    }

    private fun placeWindow() {
        val screen = windowManager.maximumWindowMetrics.bounds
        window.setLayout((screen.width() * 0.40).toInt(), WindowManager.LayoutParams.WRAP_CONTENT)
        window.attributes = window.attributes.apply {
            gravity = Gravity.TOP or Gravity.CENTER_HORIZONTAL
            y = (screen.height() * 0.15).toInt()
        }
    }

    override fun dispatchKeyEvent(event: KeyEvent): Boolean {
        if (event.keyCode == KeyEvent.KEYCODE_ESCAPE) {
            if (event.action == KeyEvent.ACTION_DOWN) close()
            return true
        }
        return super.dispatchKeyEvent(event)
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        close()
    }

    override fun onWindowFocusChanged(hasFocus: Boolean) {
        super.onWindowFocusChanged(hasFocus)
        if (hasFocus) hasHadFocus = true else if (hasHadFocus) close()
    }

    override fun onTopResumedActivityChanged(isTopResumedActivity: Boolean) {
        super.onTopResumedActivityChanged(isTopResumedActivity)
        if (!isTopResumedActivity && hasHadFocus) close()
    }

    private fun close() {
        if (isFinishing) return
        // finishAndRemoveTask, not finish: a surviving task would be brought forward on the next
        // summon and could drag the user back to the desk it was opened on.
        finishAndRemoveTask()
        overrideActivityTransition(OVERRIDE_TRANSITION_CLOSE, 0, 0)
    }
}
```

```kotlin
package dev.victorstein.reycast.core

import java.text.Normalizer

class NormalizedText(val text: String, val sourceIndex: IntArray)

object TextNormalizer {
    private val combiningMarks = Regex("\\p{Mn}+")

    fun normalize(input: String): NormalizedText {
        val text = StringBuilder()
        val sourceIndex = ArrayList<Int>()
        var i = 0
        while (i < input.length) {
            val codePoint = input.codePointAt(i)
            val folded = Normalizer.normalize(String(Character.toChars(codePoint)), Normalizer.Form.NFD)
                .replace(combiningMarks, "")
                .lowercase()
            for (c in folded) {
                text.append(c)
                sourceIndex.add(i)
            }
            i += Character.charCount(codePoint)
        }
        return NormalizedText(text.toString(), sourceIndex.toIntArray())
    }
}
```

```kotlin
package dev.victorstein.reycast.core

enum class Tier { PREFIX, WORD_START, SUBSEQUENCE }

data class Match(val tier: Tier, val gaps: Int)

object Matcher {
    private val separators = setOf(' ', '-', '_', '.')

    fun match(query: String, label: String): Match? {
        val wanted = TextNormalizer.normalize(query).text
        if (wanted.isEmpty()) return Match(Tier.PREFIX, 0)
        val normalizedLabel = TextNormalizer.normalize(label)
        val text = normalizedLabel.text
        if (text.startsWith(wanted)) return Match(Tier.PREFIX, 0)
        if (wordStarts(label, normalizedLabel).any { text.startsWith(wanted, it) }) {
            return Match(Tier.WORD_START, 0)
        }
        return subsequence(wanted, text)
    }

    private fun wordStarts(label: String, normalized: NormalizedText): List<Int> {
        val sourceStarts = HashSet<Int>()
        for (i in 1 until label.length) {
            val previous = label[i - 1]
            val current = label[i]
            val afterSeparator = previous in separators && current !in separators
            val camelHump = previous.isLowerCase() && current.isUpperCase()
            if (afterSeparator || camelHump) sourceStarts.add(i)
        }
        val index = normalized.sourceIndex
        return index.indices.filter { p ->
            index[p] in sourceStarts && (p == 0 || index[p - 1] != index[p])
        }
    }

    private fun subsequence(wanted: String, text: String): Match? {
        var from = 0
        var first = -1
        var last = -1
        for (c in wanted) {
            val found = text.indexOf(c, from)
            if (found < 0) return null
            if (first < 0) first = found
            last = found
            from = found + 1
        }
        return Match(Tier.SUBSEQUENCE, (last - first + 1) - wanted.length)
    }
}
```

```kotlin
package dev.victorstein.reycast.core

class UsageHistory {
    private val launches = HashMap<String, MutableList<Long>>()

    fun record(key: String, nowMs: Long) {
        launches.getOrPut(key) { ArrayList() }.add(nowMs)
        prune(nowMs)
    }

    fun scores(nowMs: Long): Map<String, Int> =
        launches.mapValues { (_, times) -> times.sumOf { pointsFor(nowMs - it) } }
            .filterValues { it > 0 }

    fun serialize(): String =
        launches.entries.joinToString("\n") { (key, times) -> "$key\t${times.joinToString(",")}" }

    private fun prune(nowMs: Long) {
        val iterator = launches.entries.iterator()
        while (iterator.hasNext()) {
            val entry = iterator.next()
            val kept = entry.value.filter { nowMs - it < MAX_AGE_MS }.sorted().takeLast(MAX_LAUNCHES)
            if (kept.isEmpty()) iterator.remove() else entry.setValue(kept.toMutableList())
        }
    }

    companion object {
        private const val HOUR_MS = 3_600_000L
        private const val DAY_MS = 24 * HOUR_MS
        private const val MAX_AGE_MS = 30 * DAY_MS
        private const val MAX_LAUNCHES = 20

        fun pointsFor(ageMs: Long): Int = when {
            ageMs < 4 * HOUR_MS -> 100
            ageMs < DAY_MS -> 80
            ageMs < 3 * DAY_MS -> 60
            ageMs < 7 * DAY_MS -> 40
            ageMs < MAX_AGE_MS -> 20
            else -> 0
        }

        fun parse(text: String): UsageHistory {
            val history = UsageHistory()
            for (line in text.lineSequence()) {
                val parts = line.split('\t')
                if (parts.size != 2 || parts[0].isBlank()) continue
                val times = parts[1].split(',').mapNotNull { it.toLongOrNull() }
                if (times.isNotEmpty()) history.launches[parts[0]] = times.toMutableList()
            }
            return history
        }
    }
}
```

```kotlin
package dev.victorstein.reycast.core

object Ranker {
    const val MAX_RESULTS = 8

    fun rank(entries: List<CatalogEntry>, query: String, frecency: Map<String, Int>): List<CatalogEntry> {
        val score = { entry: CatalogEntry -> frecency[entry.component] ?: 0 }
        if (query.isBlank()) {
            return entries
                .sortedWith(compareByDescending(score).thenBy { it.label.lowercase() })
                .take(MAX_RESULTS)
        }
        return entries
            .mapNotNull { entry -> Matcher.match(query.trim(), entry.label)?.let { entry to it } }
            .sortedWith(
                compareBy<Pair<CatalogEntry, Match>> { it.second.tier.ordinal }
                    .thenBy { if (it.second.tier == Tier.SUBSEQUENCE) it.second.gaps else 0 }
                    .thenByDescending { score(it.first) }
                    .thenBy { it.first.label.lowercase() },
            )
            .map { it.first }
            .take(MAX_RESULTS)
    }
}
```

```kotlin
package dev.victorstein.reycast.core

import android.view.KeyEvent

sealed interface KeyAction {
    data object Up : KeyAction
    data object Down : KeyAction
    data object Launch : KeyAction
    data object Close : KeyAction
    data class Type(val char: Char) : KeyAction
    data object PassThrough : KeyAction
}

object KeyRouter {
    /** [unicodeWithoutAlt] is the character the key produces with Alt removed from the meta state. */
    fun route(keyCode: Int, ctrl: Boolean, alt: Boolean, unicodeWithoutAlt: Int): KeyAction = when {
        keyCode == KeyEvent.KEYCODE_DPAD_UP || (ctrl && keyCode == KeyEvent.KEYCODE_K) -> KeyAction.Up
        keyCode == KeyEvent.KEYCODE_DPAD_DOWN || (ctrl && keyCode == KeyEvent.KEYCODE_J) -> KeyAction.Down
        keyCode == KeyEvent.KEYCODE_ENTER || keyCode == KeyEvent.KEYCODE_NUMPAD_ENTER -> KeyAction.Launch
        keyCode == KeyEvent.KEYCODE_ESCAPE -> KeyAction.Close
        // Cmd arrives as Alt (system Meta→Alt remap); a letter typed before Cmd is released must still type.
        alt && !ctrl && unicodeWithoutAlt > 0 && !Character.isISOControl(unicodeWithoutAlt) ->
            KeyAction.Type(unicodeWithoutAlt.toChar())
        else -> KeyAction.PassThrough
    }
}
```

```kotlin
    fun refresh(previous: List<CatalogEntry>): List<CatalogEntry> {
        val known = previous.associateBy { it.component }
        val lastUpdates = HashMap<String, Long>()
        val launcherIntent = Intent(Intent.ACTION_MAIN).addCategory(Intent.CATEGORY_LAUNCHER)
        val fresh = packageManager
            .queryIntentActivities(launcherIntent, PackageManager.ResolveInfoFlags.of(0))

                val activity = info.activityInfo
                val component = ComponentName(activity.packageName, activity.name).flattenToShortString()
                val lastUpdate = lastUpdates.getOrPut(activity.packageName) {
                    packageManager.getPackageInfo(activity.packageName, 0).lastUpdateTime

                CatalogEntry(component, label, activity.packageName, lastUpdate)

        write(fresh)
        return fresh
```

```kotlin
package dev.victorstein.reycast.android

import android.util.AtomicFile
import dev.victorstein.reycast.core.UsageHistory
import java.io.IOException
import java.util.concurrent.Executors

class UsageStore(
    private val file: AtomicFile,
    private val clock: () -> Long = System::currentTimeMillis,
) {
    private val writer = Executors.newSingleThreadExecutor()
    private val history: UsageHistory = try {
        UsageHistory.parse(String(file.readFully()))
    } catch (e: IOException) {
        UsageHistory()
    }

    fun scores(): Map<String, Int> = synchronized(history) { history.scores(clock()) }

    fun record(component: String) {
        val now = clock()
        writer.execute {
            val text = synchronized(history) {
                history.record(component, now)
                history.serialize()
            }
            val out = file.startWrite()
            try {
                out.write(text.toByteArray())
                file.finishWrite(out)
            } catch (e: IOException) {
                file.failWrite(out)
            }
        }
    }
}
```

```kotlin
    private var results: List<CatalogEntry> = emptyList()
    private var selected = 0
    private var closed = false

    init {
        val container = root.findViewById<LinearLayout>(R.id.results)
        val inflater = LayoutInflater.from(root.context)
        rows = List(Ranker.MAX_RESULTS) {
            inflater.inflate(R.layout.row_result, container, false).also(container::addView)
        }
    }

    fun open() {
        placePanel()
        root.setOnClickListener { close() }
        root.onFocusLost = ::close
        root.onKey = ::handleKey
        query.addTextChangedListener(object : TextWatcher {
            override fun beforeTextChanged(s: CharSequence?, start: Int, count: Int, after: Int) = Unit
            override fun onTextChanged(s: CharSequence?, start: Int, before: Int, count: Int) = Unit
            override fun afterTextChanged(s: Editable?) {
                selected = 0
                render()
            }
        })

        frecency = usage.scores()
        entries = catalog.cached()
        render()
        logFirstFrame()
        windowManager.addView(root, overlayParams())
        query.requestFocus()

        val cachedEntries = entries
        background.execute {
            val fresh = catalog.refresh(cachedEntries)
            main.post {
                if (!closed && fresh != entries) {
                    entries = fresh
                    render()
                }
            }
        }
    }

    fun close() {
        if (closed) return
        closed = true
        windowManager.removeView(root)
        background.shutdown()
        onClosed()
    }

    private fun placePanel() {
        val screen = windowManager.maximumWindowMetrics.bounds
        val panel = root.findViewById<View>(R.id.panel)
        panel.layoutParams = FrameLayout.LayoutParams(
            (screen.width() * 0.40).toInt(),
            FrameLayout.LayoutParams.WRAP_CONTENT,
        ).apply {
            gravity = Gravity.TOP or Gravity.CENTER_HORIZONTAL
            topMargin = (screen.height() * 0.15).toInt()
        }
    }

    private fun overlayParams() = WindowManager.LayoutParams(
        WindowManager.LayoutParams.MATCH_PARENT,
        WindowManager.LayoutParams.MATCH_PARENT,
        WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
        WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN,
        PixelFormat.TRANSLUCENT,
    )

    private fun logFirstFrame() {
        root.viewTreeObserver.addOnDrawListener(object : ViewTreeObserver.OnDrawListener {
            private var logged = false
            override fun onDraw() {
                if (logged) return
                logged = true
                Log.i(TAG, "first frame +${SystemClock.uptimeMillis() - requestedAtMs} ms")
                main.post { root.viewTreeObserver.removeOnDrawListener(this) }
            }
        })
    }

    private fun render() {
        results = Ranker.rank(entries, query.text.toString(), frecency)
        selected = selected.coerceIn(0, maxOf(0, results.lastIndex))
        rows.forEachIndexed { index, row ->
            val entry = results.getOrNull(index)
            row.visibility = if (entry == null) View.GONE else View.VISIBLE
            if (entry != null) {
                row.findViewById<TextView>(R.id.label).text = entry.label
                row.setBackgroundResource(if (index == selected) R.drawable.row_selected_bg else 0)
                bindIcon(row.findViewById(R.id.icon), entry.component)
            }
        }
    }

    private fun bindIcon(view: ImageView, component: String) {
        if (icons.containsKey(component)) {
            view.setImageDrawable(icons[component] ?: context.getDrawable(android.R.drawable.sym_def_app_icon))
            return
        }
        view.setImageDrawable(null)
        if (!iconsLoading.add(component)) return
        background.execute {
            val icon = catalog.loadIcon(component)
            main.post {
                icons[component] = icon
                if (!closed) render()
            }
        }
    }

    // Handled before the text field: it would consume ↑, move focus out on ↓ and turn Enter into an
    // editor action.
    private fun handleKey(event: KeyEvent): Boolean {
        val unicodeWithoutAlt = event.getUnicodeChar(event.metaState and KeyEvent.META_ALT_MASK.inv())
        val action = KeyRouter.route(event.keyCode, event.isCtrlPressed, event.isAltPressed, unicodeWithoutAlt)
        if (action == KeyAction.PassThrough) return false
        if (event.action == KeyEvent.ACTION_DOWN) perform(action)
        return true
    }

    private fun perform(action: KeyAction) {
        when (action) {
            KeyAction.Up -> moveSelection(-1)
            KeyAction.Down -> moveSelection(1)
            KeyAction.Launch -> results.getOrNull(selected)?.let(::launch)
            KeyAction.Close -> close()
            is KeyAction.Type -> query.text.insert(query.selectionStart.coerceAtLeast(0), action.char.toString())
            KeyAction.PassThrough -> Unit
        }
    }

    private fun moveSelection(delta: Int) {
        selected = (selected + delta).coerceIn(0, maxOf(0, results.lastIndex))
        render()
    }

    // Started while the overlay is still visible: a visible overlay window is what exempts ReyCast from
    // Android's background-activity-start block.
    private fun launch(entry: CatalogEntry) {
        val intent = Intent(Intent.ACTION_MAIN)
            .addCategory(Intent.CATEGORY_LAUNCHER)
            .setComponent(ComponentName.unflattenFromString(entry.component))
            .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_RESET_TASK_IF_NEEDED)
        try {
            context.startActivity(intent)
        } catch (e: ActivityNotFoundException) {
            return cannotOpen(entry)
        } catch (e: SecurityException) {
            return cannotOpen(entry)
        }
        usage.record(entry.component)
        close()
    }

    private fun cannotOpen(entry: CatalogEntry) {
        Toast.makeText(context, context.getString(R.string.cannot_open, entry.label), Toast.LENGTH_SHORT).show()
        entries = entries - entry
        render()
    }

    private companion object {
        const val TAG = "ReyCast"
    }
}
```
