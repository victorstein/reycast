# ReyCast Hint Mode Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Homerow-style keyboard clicking on the tablet: Alt+Shift+` shows letter badges over every visible clickable element; typing a badge's letters clicks it (Shift on the last letter long-presses).

**Architecture:** A second mode of ReyCast. The existing accessibility service gains window-content and gesture capabilities; `HintTargets` walks `getWindows()` on the background executor, `HintSession` draws badges in a non-touchable `TYPE_ACCESSIBILITY_OVERLAY` and acts with `performAction`, falling back to `dispatchGesture`. `ReyCastOverlay` owns one session of either kind. Spec: `docs/superpowers/specs/2026-11-16-hint-mode-design.md`.

**Tech Stack:** Kotlin, plain Android Views, Android 16 (compileSdk/minSdk/targetSdk 36), JUnit 4 for pure units, adb to the Galaxy Tab S8 Ultra, Key Mapper 4.7.0 (evdev triggers generated by `~/Documents/gen_evdev.py`).

```kotlin
package dev.victorstein.reycast.core

object HintLabels {
    /** Home row first, so the most common (single-letter or first) labels sit under the fingers. */
    const val ALPHABET = "sadfjklewcmpgh"

    /** Every label has the same length, which makes the set prefix-free: a full match can act at once. */
    fun assign(count: Int): List<String> {
        if (count <= 0) return emptyList()
        var length = 1
        var capacity = ALPHABET.length
        while (capacity < count) {
            length++
            capacity *= ALPHABET.length
        }
        return List(count) { index ->
            buildString {
                var remaining = index
                repeat(length) {
                    insert(0, ALPHABET[remaining % ALPHABET.length])
                    remaining /= ALPHABET.length
                }
            }
        }
    }
}
```

```kotlin
package dev.victorstein.reycast.core

class HintFilter(private val labels: List<String>) {
    var typed = ""
        private set

    /** Returns the index of the label [letter] completes; null while still partial or when ignored. */
    fun type(letter: Char): Int? {
        val next = typed + letter
        if (labels.none { it.startsWith(next) }) return null
        typed = next
        return labels.indexOf(typed).takeIf { it >= 0 }
    }

    fun backspace() {
        typed = typed.dropLast(1)
    }

    fun isVisible(index: Int) = labels[index].startsWith(typed)
}
```

```kotlin
package dev.victorstein.reycast.android

import android.accessibilityservice.AccessibilityService
import android.graphics.Rect
import android.graphics.Region
import android.view.accessibility.AccessibilityNodeInfo
import android.view.accessibility.AccessibilityWindowInfo

data class HintTarget(val node: AccessibilityNodeInfo, val rect: Rect, val editable: Boolean)

object HintTargets {
    /** Blocking cross-process calls: run off the main thread. */
    fun collect(service: AccessibilityService): List<HintTarget> {
        val targets = ArrayList<HintTarget>()
        val indexByRect = HashMap<Rect, Int>()
        // Window bounds are not cropped by windows above them, and getWindows() lists the top-most first,
        // so everything already walked covers what comes later.
        val covered = Region()
        for (window in service.windows) {
            if (window.type == AccessibilityWindowInfo.TYPE_ACCESSIBILITY_OVERLAY) continue
            val windowBounds = Rect().also(window::getBoundsInScreen)
            window.root?.let { collectWindow(it, windowBounds, covered, targets, indexByRect) }
            covered.op(windowBounds, Region.Op.UNION)
        }
        return targets
    }

    private fun collectWindow(
        root: AccessibilityNodeInfo,
        windowBounds: Rect,
        covered: Region,
        targets: MutableList<HintTarget>,
        indexByRect: MutableMap<Rect, Int>,
    ) {
        val stack = ArrayDeque<AccessibilityNodeInfo>().apply { add(root) }
        while (stack.isNotEmpty()) {
            val node = stack.removeLast()
            for (i in node.childCount - 1 downTo 0) node.getChild(i)?.let(stack::add)
            if (!node.isVisibleToUser || !node.isEnabled) continue
            if (!(node.isClickable || node.isLongClickable || node.isEditable)) continue
            val rect = Rect().also(node::getBoundsInScreen)
            if (!rect.intersect(windowBounds) || rect.isEmpty) continue
            if (covered.contains(rect.centerX(), rect.centerY())) continue
            val target = HintTarget(node, rect, node.isEditable)
            // Identical rects: the later node in tree order is the deeper one; the outer is usually a ripple wrapper.
            val existing = indexByRect[rect]
            if (existing != null) {
                targets[existing] = target
            } else {
                indexByRect[rect] = targets.size
                targets += target
            }
        }
    }
}
```

```kotlin
package dev.victorstein.reycast

import android.accessibilityservice.AccessibilityService
import android.accessibilityservice.GestureDescription
import android.graphics.Color
import android.graphics.Path
import android.graphics.PixelFormat
import android.graphics.PointF
import android.graphics.Rect
import android.graphics.Typeface
import android.os.SystemClock
import android.text.SpannableString
import android.text.Spanned
import android.text.style.ForegroundColorSpan
import android.util.Log
import android.util.TypedValue
import android.view.ContextThemeWrapper
import android.view.KeyEvent
import android.view.View
import android.view.ViewGroup
import android.view.ViewTreeObserver
import android.view.WindowManager
import android.view.accessibility.AccessibilityNodeInfo
import android.widget.FrameLayout
import android.widget.TextView
import android.widget.Toast
import dev.victorstein.reycast.android.HintTarget
import dev.victorstein.reycast.core.HintFilter
import dev.victorstein.reycast.core.HintLabels
import kotlin.math.abs

class HintSession(
    private val service: AccessibilityService,
    private val targets: List<HintTarget>,
    private val requestedAtMs: Long,
    private val onClosed: () -> Unit,
) : Session {
    private val windowManager = service.getSystemService(WindowManager::class.java)
    private val root = OverlayRoot(ContextThemeWrapper(service, android.R.style.Theme_DeviceDefault), null)
    private val labels = HintLabels.assign(targets.size)
    private val filter = HintFilter(labels)
    private val badges = labels.map(::createBadge)
    private var closed = false

    fun open() {
        root.onKey = ::handleKey
        root.onFocusLost = ::close
        badges.forEach(root::addView)
        root.viewTreeObserver.addOnGlobalLayoutListener(object : ViewTreeObserver.OnGlobalLayoutListener {
            override fun onGlobalLayout() {
                root.viewTreeObserver.removeOnGlobalLayoutListener(this)
                placeBadges()
                Log.i(TAG, "hints shown: ${targets.size} +${SystemClock.uptimeMillis() - requestedAtMs} ms")
            }
        })
        windowManager.addView(root, overlayParams())
    }

    override fun close() {
        if (closed) return
        closed = true
        try {
            if (root.isAttachedToWindow) windowManager.removeView(root)
        } finally {
            onClosed()
        }
    }

    // Not touchable: taps (the user's and the fallback gesture) reach the app beneath, and a tap there
    // moves focus away, which closes the session.
    private fun overlayParams() = WindowManager.LayoutParams(
        WindowManager.LayoutParams.MATCH_PARENT,
        WindowManager.LayoutParams.MATCH_PARENT,
        WindowManager.LayoutParams.TYPE_ACCESSIBILITY_OVERLAY,
        WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN or WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE,
        PixelFormat.TRANSLUCENT,
    ).apply {
        fitInsetsTypes = 0
        title = "ReyCast hints"
    }

    private fun createBadge(label: String) = TextView(root.context).apply {
        text = label.uppercase()
        setTextColor(Color.BLACK)
        setTextSize(TypedValue.COMPLEX_UNIT_SP, 13f)
        typeface = Typeface.create(Typeface.MONOSPACE, Typeface.BOLD)
        setBackgroundResource(R.drawable.hint_badge_bg)
        val padX = dp(4)
        val padY = dp(1)
        setPadding(padX, padY, padX, padY)
        visibility = View.INVISIBLE
        layoutParams = FrameLayout.LayoutParams(ViewGroup.LayoutParams.WRAP_CONTENT, ViewGroup.LayoutParams.WRAP_CONTENT)
    }

    private fun dp(value: Int) = (value * service.resources.displayMetrics.density).toInt()

    private fun placeBadges() {
        val origin = IntArray(2).also(root::getLocationOnScreen)
        val placed = ArrayList<PointF>()
        targets.forEachIndexed { index, target ->
            val badge = badges[index]
            val width = badge.width.toFloat()
            val height = badge.height.toFloat()
            var x = (target.rect.left - origin[0]).toFloat()
            var y = (target.rect.top - origin[1]).toFloat()
            if (placed.any { abs(it.x - x) < width && abs(it.y - y) < height }) {
                x = target.rect.exactCenterX() - origin[0] - width / 2
                y = target.rect.exactCenterY() - origin[1] - height / 2
            }
            x = x.coerceIn(0f, maxOf(0f, root.width - width))
            y = y.coerceIn(0f, maxOf(0f, root.height - height))
            badge.translationX = x
            badge.translationY = y
            placed += PointF(x, y)
        }
        render()
    }

    private fun render() {
        val typedLength = filter.typed.length
        badges.forEachIndexed { index, badge ->
            if (!filter.isVisible(index)) {
                badge.visibility = View.INVISIBLE
                return@forEachIndexed
            }
            badge.visibility = View.VISIBLE
            badge.text = SpannableString(labels[index].uppercase()).apply {
                if (typedLength > 0) {
                    setSpan(ForegroundColorSpan(TYPED_COLOR), 0, typedLength, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
                }
            }
        }
    }

    // Matched by keyCode, not by character: the summon chord's Alt may still be held, and Alt+letter
    // produces no character.
    private fun handleKey(event: KeyEvent): Boolean {
        if (closed || event.action != KeyEvent.ACTION_DOWN) return true
        when (event.keyCode) {
            KeyEvent.KEYCODE_ESCAPE -> close()
            KeyEvent.KEYCODE_DEL -> {
                filter.backspace()
                render()
            }
            in KeyEvent.KEYCODE_A..KeyEvent.KEYCODE_Z -> {
                val match = filter.type('a' + (event.keyCode - KeyEvent.KEYCODE_A))
                if (match == null) render() else act(targets[match], longPress = event.isShiftPressed)
            }
        }
        return true
    }

    private fun act(target: HintTarget, longPress: Boolean) {
        close()
        val node = target.node
        if (!node.refresh()) return toast(R.string.hint_target_gone)
        val action = if (longPress) AccessibilityNodeInfo.ACTION_LONG_CLICK else AccessibilityNodeInfo.ACTION_CLICK
        if (node.performAction(action)) {
            if (target.editable) node.performAction(AccessibilityNodeInfo.ACTION_FOCUS)
            return
        }
        val bounds = Rect().also(node::getBoundsInScreen)
        val path = Path().apply { moveTo(bounds.exactCenterX(), bounds.exactCenterY()) }
        val stroke = GestureDescription.StrokeDescription(path, 0, if (longPress) LONG_PRESS_MS else TAP_MS)
        val gesture = GestureDescription.Builder().addStroke(stroke).build()
        val dispatched = service.dispatchGesture(gesture, object : AccessibilityService.GestureResultCallback() {
            override fun onCancelled(gestureDescription: GestureDescription?) = toast(R.string.hint_cannot_click)
        }, null)
        if (!dispatched) toast(R.string.hint_cannot_click)
    }

    private fun toast(message: Int) {
        Toast.makeText(service, message, Toast.LENGTH_SHORT).show()
    }

    private companion object {
        const val TAG = "ReyCast"
        const val TAP_MS = 50L
        const val LONG_PRESS_MS = 600L
        val TYPED_COLOR = Color.argb(110, 0, 0, 0)
    }
}
```

```kotlin
object ReyCastOverlay {

        session?.close()
    }

    private val overlayContexts = HashMap<Int, Context>()

    // The overlay lands on the given display (DeX external monitor) rather than always on the tablet screen.
    fun toggle(appContext: Context, requestedAtMs: Long, display: Display? = null) {
        session?.let {
            it.close()
            if (it is OverlaySession) return
        }
        val service = accessibilityService
        val sessionContext: Context
        val windowType: Int
        if (service != null) {
            sessionContext = service
            windowType = WindowManager.LayoutParams.TYPE_ACCESSIBILITY_OVERLAY
        } else {
            if (!Settings.canDrawOverlays(appContext)) {
                Toast.makeText(appContext, R.string.needs_overlay_permission, Toast.LENGTH_LONG).show()
                return
            }
            val targetDisplay = display
                ?: appContext.getSystemService(DisplayManager::class.java).getDisplay(Display.DEFAULT_DISPLAY)
            windowType = WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY
            sessionContext = overlayContexts.getOrPut(targetDisplay.displayId) {
                appContext.createWindowContext(targetDisplay, windowType, null)
            }
        }
        val appCatalog = catalog ?: AppCatalog(appContext).also { catalog = it }
        val usageStore = usage ?: UsageStore(AtomicFile(File(appContext.filesDir, "usage.tsv")), background).also { usage = it }
        session = OverlaySession(sessionContext, appCatalog, usageStore, icons, background, requestedAtMs, windowType) { session = null }
            .also { it.open() }
    }

    fun toggleHints(appContext: Context, requestedAtMs: Long) {
        session?.let {
            it.close()
            if (it is HintSession) return
        }
        val service = accessibilityService ?: run {
            Toast.makeText(appContext, R.string.needs_accessibility_service, Toast.LENGTH_LONG).show()
            return
        }
        // shortcut: a second summon while targets are still being collected neither cancels nor reopens;
        // add a pending-collection flag if that shows up in use.
        background.execute {
            val targets = HintTargets.collect(service)
            Log.i(TAG, "hints collected: ${targets.size} +${SystemClock.uptimeMillis() - requestedAtMs} ms")
            mainHandler.post {
                if (session != null || accessibilityService !== service) return@post
                if (targets.isEmpty()) {
                    Toast.makeText(service, R.string.nothing_clickable, Toast.LENGTH_SHORT).show()
                    return@post
                }
                session = HintSession(service, targets, requestedAtMs) { session = null }.also { it.open() }
            }
        }
    }
}
```
